58 ConfigInstance::ConfigInstance(const std::string& instanceName, Role role)
59 : utils::SubCommand(!instanceName.empty() ? &utils::Config::configRoot : nullptr,
60 std::make_shared<utils::AppWithPtr>(std::string("Configuration for ")
61 .append(role == Role::SERVER ? "server" : "client")
62 .append(" instance '")
63 .append(instanceName)
64 .append("'"),
65 instanceName,
66 this),
67 "Instances")
68 , instanceName(instanceName)
69 , role(role)
70 , disableOpt(setConfigurable(addFlagFunction(
71 "--disabled{true}",
72 [this]() {
73 if (getParent() != nullptr) {
74 getParent()->disabled(this, disableOpt->as<bool>());
75 }
76 },
77 "Disable this instance",
78 "bool",
79 "false",
80 CLI::IsMember({"true", "false"})),
81 true))
82 , onDestroy([](ConfigInstance*) {
83 }) {
84 }
